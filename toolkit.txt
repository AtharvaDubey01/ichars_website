7:const BioCard = ({ title, text, boldText }: { title: string; text: string; boldText: string }) => {
8:    const [copied, setCopied] = useState(false);
9:
10:    const handleCopy = () => {
11:        navigator.clipboard.writeText(text);
12:        setCopied(true);
13:        setTimeout(() => setCopied(false), 2000);
14:    };
15:
16:    const highlightVariables = (content: string) => {
17:        const parts = content.split(/(\[[^\]]+\])/g);
18:        return parts.map((part, index) => {
19:            if (part.startsWith('[') && part.endsWith(']')) {
20:                return <span key={index} className="text-brand-deep-teal bg-brand-soft-teal/30 px-1.5 py-0.5 rounded font-medium border border-brand-deep-teal/20 mx-0.5">{part}</span>;
21:            }
22:            if (part === boldText) {
23:                return <strong key={index} className="text-brand-structural-navy font-bold">{part}</strong>;
24:            }
25:            return <span key={index}>{part}</span>;
26:        });
27:    };
28:
29:    // To handle bold, we replace boldText with exact match
30:    const withBold = text.split(boldText);
31:    const renderContent = () => {
32:        if (withBold.length === 2) {
33:             return <>{highlightVariables(withBold[0])}<strong className="text-brand-structural-navy font-bold">{boldText}</strong>{highlightVariables(withBold[1])}</>;
34:        }
35:        return highlightVariables(text);
36:    };
37:
38:    return (
39:        <Card className="bg-brand-soft-neutral/10 border-brand-soft-neutral p-6 rounded-2xl flex flex-col gap-4">
40:            <div className="flex items-center justify-between">
41:                <Heading level={4} className="text-md text-brand-structural-navy m-0">{title}</Heading>
42:                <button 
43:                    onClick={handleCopy}
44:                    className="flex items-center gap-1.5 text-xs font-semibold text-brand-deep-teal bg-white border border-brand-deep-teal/30 px-3 py-1.5 rounded-full hover:bg-brand-soft-teal/30 transition-colors shadow-sm"
45:                >
46:                    {copied ? <Check size={14} /> : <Copy size={14} />}
47:                    {copied ? 'Copied!' : 'Copy text'}
48:                </button>
49:            </div>
50:            <div className="bg-white p-5 rounded-xl border border-brand-soft-neutral/50 text-sm text-brand-secondary-text leading-relaxed shadow-inner">
51:                {renderContent()}
52:            </div>
53:        </Card>
54:    );
55:};
